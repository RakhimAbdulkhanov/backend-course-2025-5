# 1. GET - скачати картинку для коду 200 (перший раз — тягне з інтернету і зберігає в кеш)
curl -v -X GET http://localhost:3000/200 --output 200.jpg

# 2. GET - повторно отримати ту ж картинку (вже миттєво з локального кешу)
curl -v -X GET http://localhost:3000/200 --output 200_cached.jpg

# 3. PUT - закинути свій файл на сервер і зберегти як код 201 (перезаписує, якщо є)
curl -v -X PUT --data-binary @"200.jpg" http://localhost:3000/201

# 4. DELETE - видалити файл для коду 201 з кешу
curl -v -X DELETE http://localhost:3000/201

# 5. GET - перевірка неіснуючого коду (має бути 404 — не знайдено)
curl -v -X GET http://localhost:3000/999

# 6. POST - перевірка неподдержуваного методу (має бути 405 — метод не дозволений)
curl -v -X POST http://localhost:3000/200
